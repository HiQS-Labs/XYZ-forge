# ---- A5: promotion procedure step 3 on a fixture repo, using the real declared_paths -------------
import hosted_lane_publish as H
BOT = W.BOT_AUTHOR_EMAIL
def git(d, *a, env=None): return subprocess.run(["git", *a], cwd=d, capture_output=True, text=True, check=True, env=env).stdout.strip()
def step3(d, tested, ref):
    if subprocess.run(["git", "merge-base", "--is-ancestor", tested, ref], cwd=d).returncode: return False, "tested not ancestor"
    for c in git(d, "rev-list", f"{tested}..{ref}").split():
        if git(d, "show", "-s", "--format=%ae", c) != BOT: return False, f"non-bot {c[:8]}"
        paths = git(d, "diff-tree", "--no-commit-id", "--name-only", "-r", c).split()
        try: H.declared_paths(paths)
        except SystemExit as e: return False, str(e)[:60]
    return True, "ok"
base = pathlib.Path(tempfile.mkdtemp(prefix="gh1127-a5-")); up = base / "up"; up.mkdir(); git(up, "init", "-q", "-b", "development")
e = dict(os.environ, GIT_AUTHOR_NAME="n", GIT_AUTHOR_EMAIL="n@x", GIT_COMMITTER_NAME="n", GIT_COMMITTER_EMAIL="n@x")
eb = dict(e, GIT_AUTHOR_EMAIL=BOT, GIT_COMMITTER_EMAIL=BOT)
(up / "a.py").write_text("1\n"); git(up, "add", "."); git(up, "commit", "-qm", "src", env=e); tested = git(up, "rev-parse", "HEAD")
(up / "releases.sql").write_text("x\n"); rd = up / "TESTS-RESULTS/2026-10-18+GH-591" / f"wave-{tested}"; rd.mkdir(parents=True); (rd / "provenance.jsonl").write_text("{}\n")
git(up, "add", "."); git(up, "commit", "-qm", "chore: reconcile merged development work", env=eb)
ok, why = step3(up, tested, "HEAD"); check("A5: bot commit with allowlisted paths after tested -> proceed", ok, why)
