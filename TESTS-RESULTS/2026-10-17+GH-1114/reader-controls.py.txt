SHA = "a" * 40
RUN = SHA[:9] + "-4242"
def rows_of(start, suites, summary):
    out = [dict(event="run.start", **start)] + [dict(event="suite", **s) for s in suites] + [dict(event="run.summary", **summary)]
    for r in out: r.update(run=RUN, runner="validate")
    return out
def raw(rows): return "".join(json.dumps(r) + "\n" for r in rows)
def expect(label, fn, ok):
    try:
        res = fn(); got = True
    except ValueError as e:
        res = f"ValueError: {e}"; got = False
    verdict = "PASS" if got == ok else "UNEXPECTED"
    print(f"{verdict}  {label}: {'accepted' if got else 'rejected'}  {res if not got else ''}")
    assert got == ok, label
    return res

seq = rows_of(dict(commit=SHA, mode="sequential", width=1, tier=3, registered=1),
              [dict(lane="sequential", name="fixture.sh", rc=0),
               dict(lane="non-suite", name="python:test_python_layer.py", rc=0),

# (h) tier selection with a stubbed classifier
def tier_probe(tier):
    d = pathlib.Path(tempfile.mkdtemp(prefix="gh1114-tier-"))
    (d / "utils").mkdir()
    (d / "utils/ci-route.sh").write_text(f'#!/bin/sh\nSUBSYSTEM_TESTS_small="alpha.sh beta.sh"\necho tier={tier}\n')
    env = dict(os.environ, GIT_AUTHOR_NAME="t", GIT_AUTHOR_EMAIL="t@t", GIT_COMMITTER_NAME="t", GIT_COMMITTER_EMAIL="t@t")
    for c in (["git", "init", "-q"], ["git", "add", "."], ["git", "commit", "-q", "-m", "base"]):
        subprocess.run(c, cwd=d, env=env, check=True)
    (d / "x.txt").write_text("x\n"); subprocess.run(["git", "add", "."], cwd=d, env=env, check=True)
    subprocess.run(["git", "commit", "-q", "-m", "landing"], cwd=d, env=env, check=True)
    oid = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=d, text=True).strip()
    return W.select_qualification_gate(str(d), [dict(mergeCommit=dict(oid=oid), number=1)], env)
