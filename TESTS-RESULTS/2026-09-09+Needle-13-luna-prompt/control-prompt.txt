2: """Detached inference worker for the Oracle hook.
3: 
4: The Stop hook must return in well under a second or the turn is felt to hang.

7: runs the engine and writes `last/<session>.json`, which the statusline reads on

9:

11: """

13: import json, os, sys, time, warnings
14: 
15: sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
16: import oracle_config as C
17: 
18: def main() -> int:
19:     if len(sys.argv) < 2:
20:         return 0
21:     sid = sys.argv[1]
22:     pending = os.path.join(C.PENDING_DIR, f"{sid}.json")
23:     try:
24:         req = json.load(open(pending))
25:     except (OSError, ValueError):
26:         return 0
27:     cfg = C.load()
28:     os.environ.setdefault("HF_HUB_DISABLE_XET", "1"); os.environ.setdefault("NEEDLE_TELEMETRY", "0")
29:     warnings.filterwarnings("ignore")
30:     t0 = time.perf_counter()
31:     rec = {"session_id": sid, "prompt_id": req.get("prompt_id"), "ts": req.get("ts"),
32:            "history_len": req.get("history_len"), "query": req.get("query"),
33:            "cact": os.path.basename(cfg["cact"]), "recommendations": [], "reasoning": None,
34:            "engine": {}, "ms": None}

36:         import needle
37:         agent = needle.Needle(tools=C.tools_json(cfg["labels"]), system=C.SYSTEM, weights=cfg["cact"])

40:         finally:
41:             try: agent.close()
42:             except Exception: pass

44:         calls = out.get("function_calls") or []
45:         # v0 is top-1: the engine returns one call or an abstention ([]). The
46:         # templated why is the label's own description -- never generated prose.
47:         rec["recommendations"] = [{"label": c["name"], "why": desc.get(c["name"], "")}
48:                                   for c in calls if c.get("name") in desc][:3]

51:     except Exception as exc:
52:         rec["error"] = f"{type(exc).__name__}: {exc}"[:300]

59:     try: os.remove(pending)
60:     except OSError: pass
61:     # PR #4's log, now with the recommendations field filled in.
62:     with open(C.HOOK_LOG, "a") as fh:
63:         fh.write(json.dumps({"ts": rec["ts"], "session_id": sid, "prompt_id": rec["prompt_id"],
64:                              "query": rec["query"], "recommendations": rec["recommendations"],
65:                              "ms": rec["ms"], "cact": rec["cact"]}, ensure_ascii=False) + "\n")
66:     return 0
67:

69:     raise SystemExit(main())

1: """The ONE `query` serializer shared by training and the end-of-turn hook (#1 §3, §6).
2: 
3: #1 §6: "A mismatch between hook-time and training-time serialization is the single
4: most likely cause of a good model looking useless in production -- build them from one
5: shared function, not two implementations." This module is that function. The corpus
6: builder (`build_oracle_jsonl.py`) and the Stop hook (`utils/hooks/oracle_stop_hook.py`)
7: both call `serialize_query`; neither has a format of its own. `tests/test_serialize.py`
8: asserts the two paths produce byte-identical queries for the same steps.
9: 
10: FORMAT `q1` -- designed so `reasoning` can cite a span inside it (#1 §3)
11: 
12:     [oracle-q1]
13:     REQUEST: <most recent user request, truncated>
14:     RECENT ACTIONS (oldest->newest): read_file, search_code, git_inspect
15:     LAST: git_inspect
16: 
17: Every line is a citable anchor: `REQUEST:`, `RECENT ACTIONS`, `LAST:`. The templated
18: reasoning cites `LAST:`; a future richer reasoning can cite `REQUEST:` spans. Labels are
19: v1 names from `taxonomy.LABELS_V1`, never raw tool names, so the model sees the same
20: vocabulary it must emit. The `[oracle-q1]` marker is a version stamp: if the hook and
21: the trainer ever disagree on format, the mismatch is detectable in the data rather than
22: discovered as a silently bad model.
23: 
24: TOKEN BUDGET -- measured, not assumed (2026-09-07, real Needle tokenizer)
25: 
26:     44 full label schemas   1,383 tokens   <-- exceeds finetune's default --max-len 1024 alone
27:     name+description only   1,080 tokens
28:     names only                435 tokens
29:     sample q1 query            83 tokens
30: 
31: `finetune._encode` truncates `ids[:max_len]` from the END, i.e. the TARGET is cut first
32: and the row trains toward nothing, silently. So the schemas are kept in full (the shape
33: the base model was trained on) and training runs at `--max-len 2048`, the architecture's
34: `max_seq_len`. `build_oracle_jsonl.py --check-max-len` refuses any row that would
35: truncate. Do not shrink schemas to fit 1024; raise the cap instead.
36: """

128: 1. **Widen readers.** Loaders grow an explicit accept-set (e.g. `{"@1", "@2"}`). Old
129:    artifacts keep working; new ones become readable. No producer changes yet.
130: 2. **Flip producers.** Producers stamp `@2`. Because every reader in the repo already
131:    accepts `@2`, the flip cannot orphan a consumer.
132: 3. **Narrow readers** — no sooner than the next release cycle after step 2 — to `{"@2"}`
133:    only. This is the deprecation window's far edge: bounded by release cycles, not by
134:    "whenever," and it lands only when the golden fixtures and the hostile-input matrix have
135:    both run against the surviving version.

138: type changes, and semantic redefinitions of an existing field require a new contract name,
139: not a version bump — a consumer must be able to read `@N+1` with its `@N` logic plus
140: null-tolerant accessors for fields it doesn't know.
