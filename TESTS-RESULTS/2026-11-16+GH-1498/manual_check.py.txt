# GH-1498 recorded manual check (NOT a test suite; GH-831). Run: python3 manual_check.py.txt <repo-root>
# A stateful fake `gh` on PATH serves canned PRs/history and logs every write verb. Each assertion has a
# pinned expected outcome and a controlled bad candidate (an in-process monkeypatch, discarded after the
# run, never committed) that must make the same assertion fail.

NOW = datetime.now(timezone.utc)
iso = lambda d: (NOW - timedelta(days=d)).isoformat().replace("+00:00", "Z")
RESULTS = []
ORIG_PATH = os.environ["PATH"]

def commit(days, headline="work", parents=1, email="dev@example.com", oid=None):
    return {"commit": {"oid": oid or f"c{days}{headline[:3]}{parents}", "committedDate": iso(days),
                       "messageHeadline": headline, "committer": {"email": email},
                       "parents": {"totalCount": parents}}}


def conn(nodes, more=False):
    return {"pageInfo": {"hasPreviousPage": more}, "nodes": nodes}


def history(commits=(), comments=(), more_commits=False):
    return {"commits": conn(list(commits), more_commits), "comments": conn(list(comments)),
            "reviews": conn([]), "reviewThreads": conn([])}


def mkpr(n, age, cond="clean", **kw):
    p = {"number": n, "state": "OPEN", "title": kw.get("title", f"feat: pr {n}"), "body": kw.get("body", ""),
         "labels": [{"name": l} for l in kw.get("labels", [])], "isDraft": kw.get("draft", False),
         "headRefName": kw.get("head", f"feat/pr{n}"), "baseRefName": kw.get("base", "development"),
         "mergeable": "CONFLICTING" if cond == "dirty" else "MERGEABLE",
         "mergeStateStatus": {"dirty": "DIRTY", "behind": "BEHIND"}.get(cond, "CLEAN"),
         "statusCheckRollup": ([{"name": "build", "conclusion": "FAILURE", "detailsUrl": kw.get("url", "https://ci/1")}]
                               if cond == "failing" else []),
         "headRefOid": f"h{n}", "isCrossRepository": False, "maintainerCanModify": True,
         "createdAt": iso(60), "updatedAt": iso(kw.get("updated", age)), "url": f"u{n}",
         "reviewDecision": "", "reviews": [], "files": [], "author": {"login": "x"}, "_age": age}
    return p


class Env:
    def __init__(self, prs, activity=None, **extra):
        self.dir = Path(tempfile.mkdtemp(prefix="gh1498-"))
        (self.dir / "bin").mkdir(); (self.dir / "repo").mkdir(); (self.dir / "state").mkdir()
        gh = self.dir / "bin/gh"; gh.write_text(FAKE_GH); gh.chmod(gh.stat().st_mode | stat.S_IEXEC)
        acts = activity or {str(p["number"]): history([commit(p["_age"])]) for p in prs}
        self.state = {"slug": "acme/repo", "prs": prs, "activity": acts, "comments": {}, **extra}
        self.save()
        os.environ.update(FAKE_GH_STATE=str(self.dir / "st.json"), FAKE_GH_LOG=str(self.dir / "writes.log"),
                          PR_MINDER_STATE=str(self.dir / "state"),
                          PATH=f"{self.dir / 'bin'}:{ORIG_PATH}")

    def save(self):
        (self.dir / "st.json").write_text(json.dumps(self.state))

    def load(self):
        return json.loads((self.dir / "st.json").read_text())

    def writes(self):
        f = self.dir / "writes.log"
        return [json.loads(l) for l in f.read_text().splitlines()] if f.exists() else []

    def receipts(self):
        f = self.dir / "state/acme/repo/actions.jsonl"
        return [json.loads(l) for l in f.read_text().splitlines()] if f.exists() else []

import pr_minder as pm  # noqa: E402
pc, mc = pm.pc, pm.mc


def run(*argv):
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf), contextlib.redirect_stderr(io.StringIO()):
        pm.main(["--repo-root", os.environ.get("REPO_UNDER", "."), *argv])
    return buf.getvalue()


def plans(env, *argv):
    os.environ["REPO_UNDER"] = str(env.dir / "repo")
    return {r["pr"]: r for r in json.loads(run("--json", *argv))}


def apply(env):
    os.environ["REPO_UNDER"] = str(env.dir / "repo")
    run("--apply")
    return env.writes()


@contextlib.contextmanager
def patched(obj, name, value):
    old = getattr(obj, name); setattr(obj, name, value)
    try:
        yield
    finally:
        setattr(obj, name, old)


def check(aid, desc, good, bad_desc, bad):
    """good/bad are callables returning bool (assertion holds). The bad candidate must FAIL the assertion."""
    try:
        g = bool(good())
    except Exception as exc:
        g = False; desc += f" [good raised {exc!r}]"
    try:
        b = bool(bad()) if bad else None
    except Exception:
        b = False
    ok = g and (b is None or b is False)
    RESULTS.append({"id": aid, "assertion": desc, "candidate": "PASS" if g else "FAIL",
                    "bad_candidate": bad_desc, "bad_rejected": None if b is None else (b is False),
                    "verdict": "PASS" if ok else "FAIL"})
    print(f"{'PASS' if ok else 'FAIL'} {aid}: {desc} | bad: {bad_desc} -> "
          f"{'n/a' if b is None else ('rejected' if b is False else 'NOT rejected')}")


def acts(row):
    p = row["plan"]
    return (bool(p["update_branch"]), p["label"] if p["label"] not in (None,) else None, bool(p["comment"]))


# ---------------- A1 zero writes ----------------
def a1():

    e = Env(prs); plans(e); good = len(e.writes()) == 0

check("A1", "plan mode logs zero write verbs", lambda: a1()[0], "--apply against the same shim", a1_bad)

# ---------------- A2 rung matrix ----------------
EXPECT = {  # (age, cond) -> (update_branch, label, comment)
    (1.5, "behind"): (False, None, False), (3.1, "behind"): (True, None, False),
    (7.1, "behind"): (True, "pr-minder:stale", False), (14.1, "behind"): (False, "pr-minder:decide", True),
    (30.1, "behind"): (False, "keep", False),
    (1.5, "dirty"): (False, None, False), (3.1, "dirty"): (False, None, True),
    (7.1, "dirty"): (False, "pr-minder:stale", True), (14.1, "dirty"): (False, "pr-minder:decide", True),
    (30.1, "dirty"): (False, "keep", False),
    (1.5, "failing"): (False, None, False), (3.1, "failing"): (False, None, True),
    (7.1, "failing"): (False, "pr-minder:stale", True), (14.1, "failing"): (False, "pr-minder:decide", True),
    (30.1, "failing"): (False, "keep", False),
}


def a2_rows():
    prs, key = [], {}
    for i, (age, cond) in enumerate(EXPECT, start=10):
        prs.append(mkpr(i, age, cond)); key[i] = (age, cond)
    prs.append(mkpr(90, 8.1, "behind", draft=True))            # draft floors +4 -> rung 1, no update-branch
    prs.append(mkpr(91, 4.1, "behind"))                        # drift 50 > 30 -> rung 2
    e = Env(prs, behind={"91": 50})
    return plans(e), key


def a2_holds():
    rows, key = a2_rows()
    for n, k in key.items():
        if acts(rows[n]) != EXPECT[k]:
            return False
    return (rows[90]["rung"] == 1 and not rows[90]["plan"]["update_branch"] and rows[90]["plan"]["comment"]
            and rows[91]["rung"] == 2)


def off_by_one(age, draft, stale, draft_stale):
    shift = (draft_stale - stale) if draft else 0
    return sum(1 for f in [stale + shift, 8 + shift, 15 + shift, 31 + shift] if age >= f)


def a2_bad():
    with patched(pm, "base_rung", off_by_one):
        return a2_holds()


check("A2", "each rung x {behind,dirty,failing} + draft + drift-bump gives the ladder's exact bundle",
      a2_holds, "rung floors off by one day (8/15/31)", a2_bad)

# ---------------- A4 idempotency ----------------
def a4(bad=False):

def planner_report(env, patch=None):
    code = ("import sys; sys.path.insert(0, %r); import planner_core as pc\n" % str(PLANNER)) + (patch or "") + \
           "\nsys.argv=['planner_core.py','--mode','stalled','--repo-root',%r,'--gh-repo','acme/repo'," \
           "'--local-roots',%r,'--operator','op','--device-id','dev']\npc.main()" % (str(env.dir / "repo"), str(env.dir / "repo"))
    subprocess.run([sys.executable, "-c", code], capture_output=True, text=True, env=os.environ.copy())
    js = next((env.dir / "repo/temp/planner").glob("stalled-*.json"))
    return json.loads(js.read_text())["stalled_stale"]


DROP = ("_orig = pc.collect_stalled_stale\n"
        "def _drop(*a, **k):\n    k.pop('activity', None); return _orig(*a, **k)\n"
        "pc.collect_stalled_stale = _drop")


def a6_ok():
    sec = planner_report(a5_env())
    row = next(r for r in sec["stalled"] if r["ref"] == "PR #40")
    err = planner_report(a5_env(graphql_error=True))
    return (row["last_activity"] == iso(20)[:10]
            and any("#40" in d and "updatedAt was used" in d for d in err["disclosures"]))


def a6_bad():
    sec = planner_report(a5_env(), DROP)
    row = next((r for r in sec["stalled"] if r["ref"] == "PR #40"), None)
    return bool(row) and row["last_activity"] == iso(20)[:10]


check("A6", "real `planner_core.py --mode stalled` report shows the 20d-old date; history error -> report names #40 as fallback",
      a6_ok, "production call without the activity pass-through", a6_bad)

# ---------------- A10 receipts ----------------
def paired(rows):
    intents = [r for r in rows if r.get("phase") == "intent"]
    results = [r for r in rows if r.get("phase") == "result" and r.get("action") != "bundle"]
    return len(intents) == len(results) > 0 and all(
        any(x["pr"] == i["pr"] and x["action"] == i["action"] for x in results) for i in intents)


def a10_ok():
    e = Env([mkpr(80, 4, "behind"), mkpr(81, 8, "dirty")]); apply(e)
    f = Env([mkpr(82, 4, "behind")], fail_update=True); apply(f)
    return paired(e.receipts()) and any(r.get("outcome") == "failed" for r in f.receipts())


def a10_bad():
    orig = pm.Receipts.write

    def drop_failed(self, **row):
        if row.get("outcome") == "failed":
            return
        return orig(self, **row)
    with patched(pm.Receipts, "write", drop_failed):
        f = Env([mkpr(82, 4, "behind")], fail_update=True); apply(f)
    return paired(f.receipts())


check("A10", "every write has intent+result; a forced update-branch failure records result: failed", a10_ok,
      "result line skipped on failure", a10_bad)
