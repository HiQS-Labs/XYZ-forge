# run, never committed) that must make the same assertion fail.

def commit(days, headline="work", parents=1, email="dev@example.com", oid=None):
    return {"commit": {"oid": oid or f"c{days}{headline[:3]}{parents}", "committedDate": iso(days),
                       "messageHeadline": headline, "committer": {"email": email},
                       "parents": {"totalCount": parents}}}


def conn(nodes, more=False):
    return {"pageInfo": {"hasPreviousPage": more}, "nodes": nodes}


def history(commits=(), comments=(), more_commits=False):
    return {"commits": conn(list(commits), more_commits), "comments": conn(list(comments)),
            "reviews": conn([]), "reviewThreads": conn([])}


def mkpr(n, age, cond="clean", **kw):
    p = {"number": n, "state": "OPEN", "title": kw.get("title", f"feat: pr {n}"), "body": kw.get("body", ""),
         "labels": [{"name": l} for l in kw.get("labels", [])], "isDraft": kw.get("draft", False),
         "headRefName": kw.get("head", f"feat/pr{n}"), "baseRefName": kw.get("base", "development"),
         "mergeable": "CONFLICTING" if cond == "dirty" else "MERGEABLE",
         "mergeStateStatus": {"dirty": "DIRTY", "behind": "BEHIND"}.get(cond, "CLEAN"),
         "statusCheckRollup": ([{"name": "build", "conclusion": "FAILURE", "detailsUrl": kw.get("url", "https://ci/1")}]
                               if cond == "failing" else []),
         "headRefOid": f"h{n}", "isCrossRepository": False, "maintainerCanModify": True,
         "createdAt": iso(60), "updatedAt": iso(kw.get("updated", age)), "url": f"u{n}",
         "reviewDecision": "", "reviews": [], "files": [], "author": {"login": "x"}, "_age": age}
    return p


class Env:
    def __init__(self, prs, activity=None, **extra):
        self.dir = Path(tempfile.mkdtemp(prefix="gh1498-"))
        (self.dir / "bin").mkdir(); (self.dir / "repo").mkdir(); (self.dir / "state").mkdir()
        gh = self.dir / "bin/gh"; gh.write_text(FAKE_GH); gh.chmod(gh.stat().st_mode | stat.S_IEXEC)
        acts = activity or {str(p["number"]): history([commit(p["_age"])]) for p in prs}
        self.state = {"slug": "acme/repo", "prs": prs, "activity": acts, "comments": {}, **extra}
        self.save()
        self.activate()

    def activate(self):
        """Point the fake gh, its write log and pr-minder's receipts at THIS env (several envs may coexist)."""
        os.environ.update(FAKE_GH_STATE=str(self.dir / "st.json"), FAKE_GH_LOG=str(self.dir / "writes.log"),
                          PR_MINDER_STATE=str(self.dir / "state"), REPO_UNDER=str(self.dir / "repo"),
                          PATH=f"{self.dir / 'bin'}:{ORIG_PATH}")

    def save(self):
        (self.dir / "st.json").write_text(json.dumps(self.state))

    def load(self):
        return json.loads((self.dir / "st.json").read_text())

    def writes(self):
        f = self.dir / "writes.log"
        return [json.loads(l) for l in f.read_text().splitlines()] if f.exists() else []

    def receipts(self):
        f = self.dir / "state/acme/repo/actions.jsonl"
        return [json.loads(l) for l in f.read_text().splitlines()] if f.exists() else []


def own(env, n, label="pr-minder:stale"):
    """Seed a receipt proving pr-minder added `label` to PR n (ownership for removal)."""
    rec = env.dir / "state/acme/repo"; rec.mkdir(parents=True, exist_ok=True)
    with (rec / "actions.jsonl").open("a") as fh:
        fh.write(json.dumps({"ts": iso(30), "pr": n, "phase": "result", "action": "add-label",
                             "arg": label, "outcome": "ok"}) + "\n")
    return env

def run(*argv):
    buf = io.StringIO()
    with contextlib.redirect_stdout(buf), contextlib.redirect_stderr(io.StringIO()):
        pm.main(["--repo-root", os.environ.get("REPO_UNDER", "."), *argv])
    return buf.getvalue()


def plans(env, *argv):
    env.activate()
    return {r["pr"]: r for r in json.loads(run("--json", *argv))}


def apply(env):
    env.activate()
    run("--apply")
    return env.writes()


@contextlib.contextmanager
def patched(obj, name, value):
    old = getattr(obj, name); setattr(obj, name, value)
    try:
        yield
    finally:
        setattr(obj, name, old)


def check(aid, desc, good, bad_desc, bad):
    """good/bad are callables returning bool (assertion holds). The bad candidate must FAIL the assertion."""
    try:
        g = bool(good())
    except Exception as exc:
        g = False; desc += f" [good raised {exc!r}]"
    try:
        b = bool(bad()) if bad else None
    except Exception:
        b = False
    ok = g and (b is None or b is False)
    RESULTS.append({"id": aid, "assertion": desc, "candidate": "PASS" if g else "FAIL",
                    "bad_candidate": bad_desc, "bad_rejected": None if b is None else (b is False),
                    "verdict": "PASS" if ok else "FAIL"})
    print(f"{'PASS' if ok else 'FAIL'} {aid}: {desc} | bad: {bad_desc} -> "
          f"{'n/a' if b is None else ('rejected' if b is False else 'NOT rejected')}")


def acts(row):

    return (bool(p["update_branch"]), p["label"] if p["label"] not in (None,) else None, bool(p["comment"]))

def a2_rows():

    for i, (age, cond) in enumerate(EXPECT, start=10):

    prs.append(mkpr(90, 8.1, "behind", draft=True))            # draft floors +4 -> rung 1, no update-branch

    return (all(rows[n]["rung"] == 4 and not rows[n]["plan"]["write"] for n in rung4)
            and rows[90]["rung"] == 1 and not rows[90]["plan"]["update_branch"] and rows[90]["plan"]["comment"]
            and rows[91]["rung"] == 2)


def off_by_one(age, draft, stale, draft_stale):
    shift = (draft_stale - stale) if draft else 0
    return sum(1 for f in [stale + shift, 8 + shift, 15 + shift, 31 + shift] if age >= f)


def a2_bad():
    with patched(pm, "base_rung", off_by_one):
        return a2_holds()


check("A2", "each rung x {behind,dirty,failing} + draft + drift-bump gives the ladder's exact bundle",
      a2_holds, "rung floors off by one day (8/15/31)", a2_bad)

def a8_bad_names():
    """A predicate bound to the PR's CURRENT head name: right for the original name, wrong after a rename."""
    def bound(head):
        def current_names(commit, receipt_oids=frozenset()):
            return ((commit.get("parents") or {}).get("totalCount") == 2
                    and commit.get("messageHeadline") == f"Merge branch 'development' into {head}")
        return current_names
    with patched(pc, "is_maintenance_commit", bound("fix/x")):

    with patched(pc, "is_maintenance_commit", bound("fix/new")):
        renamed = plans(a8_env("rename-head"))[60]["rung"] == 2        # same OID, head renamed
    return original_ok and renamed

# ---------------- A10 receipts ----------------
def paired(rows):
    intents = [r for r in rows if r.get("phase") == "intent"]
    results = [r for r in rows if r.get("phase") == "result" and r.get("action") != "bundle"]
    return len(intents) == len(results) > 0 and all(
        any(x["pr"] == i["pr"] and x["action"] == i["action"] for x in results) for i in intents)


def write_results(rows):
    return [r for r in rows if r.get("phase") == "result" and r.get("action") != "bundle"]


def a10_ok():

    f = Env([mkpr(82, 4, "behind")], fail_update=True); apply(f)

    return (paired(e.receipts()) and len(write_results(e.receipts())) == len(w)       # 1 receipt per shim write
            and any(r.get("action") == "label-create" for r in e.receipts())
