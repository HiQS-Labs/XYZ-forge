Usage: python3 gh1452-manual-check.py <gate-clone-root>

def ev(label, **kv):
    print("EVIDENCE " + label + ": " + json.dumps(kv, sort_keys=True, default=str))


def ins(path, text):
    def f(root):
        p = root / path
        lines = p.read_text().splitlines(keepends=True)
        lines.insert(1, text + "\n")
        p.write_text("".join(lines))
    return f


def both(*fns):
    def f(root):
        for fn in fns:
            fn(root)
    return f


def newfile(name):
    return lambda root: (root / name).write_text(f"{name}\n")


class _Pool:
    pass


def _snap_writer(args):
    path, tag, rounds = args
    for i in range(rounds):
        pins = {str(n): f"{tag}{n:02d}{i:04d}" + "0" * 30 for n in range(1, 4)}
        merge_cleanup.pin_selection.__globals__["selection_path"]  # name resolves in the real module
        fc = {"version": 1, "order": [1, 2, 3], "prs": {n: {"head": h} for n, h in pins.items()}, "recommendation": "group"}
        open_prs = [{"number": int(n), "headRefOid": h, "isDraft": False} for n, h in pins.items()]
        with mock.patch.object(merge_cleanup, "selection_path", return_value=Path(path)), \
             mock.patch.object(merge_cleanup, "origin_url", return_value="x"), \
             contextlib.redirect_stdout(io.StringIO()):
            assert merge_cleanup.pin_selection(Path("/nonexistent"), DEV, {1: None, 2: None, 3: None}, open_prs, forecast=fc)


class Manual(B.LedgerFixture):
    def setUp(self):
        super().setUp()
        (self.seed / "CHANGELOG.md").write_text("# Changelog\n\n## 2026-11-01\n- base\n")
        (self.seed / "app.py").write_text("".join(f"line {i}\n" for i in range(1, 11)))
        B.commit_all(self.seed, "add changelog and app")
        B._git(self.seed, "push", "-q", "origin", DEV)
        for c in (self.primary, self.probe):
            B._git(c, "pull", "-q", "--ff-only")

    # --- helpers ---
    def pr(self, n, fn, pull=True):
        self.branch(f"pr{n}", n, fn)
        if pull:
            B._git(self.probe, "fetch", "-q", "origin")
            B._git(self.probe, "push", "-q", "origin", f"origin/pr{n}:refs/pull/{n}/head")

    def advance(self, fn):
        fn(self.seed)
        B.commit_all(self.seed, "advance development")
        B._git(self.seed, "push", "-q", "origin", DEV)
        B._git(self.primary, "fetch", "-q", "origin")

    def forecast(self, nums, mutate=None):
        prs = merge_cleanup.fetch_open_prs_with_retry(str(self.primary))
        by = {int(p["number"]): p for p in prs}
        ordered, _, _ = merge_cleanup.toposort_prs([by[n] for n in nums])
        if mutate:
            mutate(ordered)
        return merge_cleanup.build_landing_forecast(self.primary, DEV, ordered)

    def ledger_conflicting_queue(self):
        for n in (1, 2, 3):
            self.pr(n, both(ins("releases.sql", f"-- pr{n} note"), ins("CHANGELOG.md", f"- pr{n} entry")))
        self.advance(both(ins("releases.sql", "-- landed note"), ins("CHANGELOG.md", "- landed entry")))

    def sel(self):
        return merge_cleanup.selection_path(self.primary.resolve(), merge_cleanup.origin_url(self.primary) or "", DEV)

    def cli(self, execute, extra):
        out = io.StringIO()
        with contextlib.redirect_stdout(out):
            rc = self.run_main(execute=execute, extra=extra)
        return rc, out.getvalue()

    def mutating_calls(self):
        return [c for c in self.load().get("calls", []) if c[:2] in (["issue", "create"], ["issue", "edit"], ["pr", "merge"])]

    def batch_files(self):
        root = self.primary / ".tick" / "merge-cleanup"
        return sorted(str(p.relative_to(root)) for p in root.rglob("batch-*.json")) if root.exists() else []

    def test_01_three_conflicting_ledger_prs_forecast_group(self):
        self.ledger_conflicting_queue()
        fc = self.forecast([1, 2, 3])
        text = merge_cleanup.format_landing_forecast(fc, DEV)
        print(text)
        ev("S1 group forecast", recommendation=fc["recommendation"], reason=fc["reason"], deferrals=fc["repair_deferrals"],
           seq=[fc["prs"][str(n)]["sequence"]["state"] for n in (1, 2, 3)],
           eligible=fc["prs"]["1"]["alone"]["b1_eligible"], review=fc["prs"]["1"]["alone"]["needs_review"])
        self.assertEqual(fc["recommendation"], "group")
        self.assertEqual([fc["prs"][str(n)]["sequence"]["state"] for n in (1, 2, 3)], ["conflict", "conditional", "conditional"])
        self.assertEqual(sorted(fc["prs"]["1"]["alone"]["b1_eligible"]), ["CHANGELOG.md", "releases.sql"])
        self.assertEqual(fc["prs"]["1"]["alone"]["needs_review"], [])
        # RED CONTROL: break the classifier (every merge-tree reads clean) and the same queue must stop saying group.
        with mock.patch.object(ledger_merge, "_merge_tree", return_value=(0, "t" * 40, set(), "")):
            wrong = self.forecast([1, 2, 3])
        ev("S1 red control (classifier broken)", recommendation=wrong["recommendation"])
        self.assertNotEqual(wrong["recommendation"], "group")

    def test_09_snapshot_compat_and_comparison(self):
        self.ledger_conflicting_queue()
        fc = self.forecast([1, 2, 3])
        prs = merge_cleanup.fetch_open_prs_with_retry(str(self.primary))
        self.assertTrue(merge_cleanup.pin_selection(self.primary, DEV, {1: None, 2: None, 3: None}, prs, forecast=fc))
        data = json.loads(self.sel().read_text())
        ev("S9 snapshot", keys=sorted(data), forecast_heads_equal_pins={n: data["forecast"]["prs"][n]["head"] == h for n, h in data["prs"].items()})
        self.assertEqual({n: e["head"] for n, e in data["forecast"]["prs"].items()}, data["prs"])
        verified = merge_cleanup.verify_selection(self.primary, DEV, {1: None, 2: None, 3: None}, prs)
        self.assertEqual({str(n): h for n, h in verified.items()}, data["prs"])
        order = fc["order"]
        pins = {n: h for n, h in verified.items()}
        for decision, expect in (("group", "match"), ("direct", "override")):
            cmp_ = merge_cleanup.forecast_comparison(self.primary, DEV, decision, pins, order)
            ev("S9 comparison", decision=decision, state=cmp_["state"], stale=cmp_["stale"])
            self.assertEqual(cmp_["state"], expect)
            self.assertFalse(cmp_["stale"], cmp_)
        wrong_order = merge_cleanup.forecast_comparison(self.primary, DEV, "group", pins, list(reversed(order)))
        wrong_heads = merge_cleanup.forecast_comparison(self.primary, DEV, "group", {**pins, 1: "f" * 40}, order)
        wrong_set = merge_cleanup.forecast_comparison(self.primary, DEV, "group", {1: pins[1], 2: pins[2]}, order[:2])
        ev("S9 stale reasons", order=wrong_order["stale_reasons"], heads=wrong_heads["stale_reasons"], membership=wrong_set["stale_reasons"])
        self.assertTrue(wrong_order["stale"] and wrong_heads["stale"] and wrong_set["stale"])
        self.advance(ins("app.py", "base moves"))
        moved = merge_cleanup.forecast_comparison(self.primary, DEV, "group", pins, order)
        ev("S9 base moved", stale_reasons=moved["stale_reasons"])
        self.assertTrue(any("base moved" in r for r in moved["stale_reasons"]))
        # a pin file written by the old code (no forecast key) still verifies, and compares as None
        self.sel().write_text(json.dumps({"integration": DEV, "prs": data["prs"]}))
        self.assertIsNotNone(merge_cleanup.verify_selection(self.primary, DEV, {1: None, 2: None, 3: None}, prs))
        self.assertIsNone(merge_cleanup.forecast_comparison(self.primary, DEV, "group", pins, order))
        ev("S9 old-format pin", verifies=True, comparison=None)

    def test_10_concurrent_snapshots_stay_coherent_and_naive_rmw_does_not(self):
        path = self.tmp / "sel.json"
        ctx = multiprocessing.get_context("fork")
        procs = [ctx.Process(target=_snap_writer, args=((str(path), tag, 40),)) for tag in ("a", "b")]
        for p in procs:
            p.start()
        bad, reads = 0, 0
        while any(p.is_alive() for p in procs):
            try:
                d = json.loads(path.read_text())
            except (OSError, ValueError):
                if path.exists():
                    bad += 1
                continue
            reads += 1
            if {n: e["head"] for n, e in d["forecast"]["prs"].items()} != d["prs"]:
                bad += 1
        for p in procs:
            p.join()
        d = json.loads(path.read_text())
        final_ok = {n: e["head"] for n, e in d["forecast"]["prs"].items()} == d["prs"]
        ev("S10 concurrent snapshots", reads=reads, incoherent_or_torn_reads=bad, final_coherent=final_ok, exit_codes=[p.exitcode for p in procs])
        self.assertEqual(bad, 0)
        self.assertTrue(final_ok)
        self.assertEqual([p.exitcode for p in procs], [0, 0])
        # RED CONTROL: the naive pattern (write pins, later read-modify-write the forecast in) is detectably wrong.
        naive = self.tmp / "naive.json"
        naive.write_text(json.dumps({"prs": {"1": "A" * 40}}))                 # run A pins
        naive.write_text(json.dumps({"prs": {"1": "B" * 40}}))                 # run B replaces the pin while A forecasts
        d = json.loads(naive.read_text())
        d["forecast"] = {"prs": {"1": {"head": "A" * 40}}}                      # run A attaches its forecast to B's pins
        naive.write_text(json.dumps(d))
        d = json.loads(naive.read_text())
        mismatch = {n: e["head"] for n, e in d["forecast"]["prs"].items()} != d["prs"]
        ev("S10 red control (naive read-modify-write)", detected_mismatch=mismatch)
        self.assertTrue(mismatch)

    def test_11_cli_dry_run_modes_and_execute_refusal(self):
        self.ledger_conflicting_queue()
        # fewer than 3 listed PRs: no forecast, no forecast key
        rc, out = self.cli(False, ["--prs", "1,2"])
        two = json.loads(self.sel().read_text())
        ev("S11 two PRs", rc=rc, forecast_key="forecast" in two, printed="Landing forecast (GH-1452" in out)
        ev("S11 two PRs stderr tail", err=[l for l in self.err.splitlines() if "ERROR" in l or "WARNING" in l][-4:])
        self.assertNotIn("forecast", two)
        self.assertNotIn("Landing forecast (GH-1452", out)
        # three PRs: forecast printed and persisted with the pins
        rc, out = self.cli(False, ["--prs", "1,2,3"])
        three = json.loads(self.sel().read_text())
        ev("S11 three PRs", rc=rc, printed="Landing forecast (GH-1452" in out, recommendation_line=[l for l in out.splitlines() if l.startswith("Recommendation:")],
           advice="--landing-plan group --landing-reason" in out, persisted=three["forecast"]["recommendation"])
        self.assertIn("Landing forecast (GH-1452", out)
        self.assertEqual(three["forecast"]["recommendation"], "group")
        self.assertIn("Forecast advice, not a decision", out)
        # --scan-only and a dry run without --prs leave the stored snapshot byte-identical
        before = self.sel().read_bytes()
        self.cli(False, ["--scan-only"])
        self.cli(False, [])
        ev("S11 scan-only / no --prs", unchanged=self.sel().read_bytes() == before)
        self.assertEqual(self.sel().read_bytes(), before)
        # --prs-only dry run forecasts
        self.cli(False, ["--prs", "1,2,3", "--prs-only"])
        ev("S11 prs-only", forecast="forecast" in json.loads(self.sel().read_text()))
        self.assertIn("forecast", json.loads(self.sel().read_text()))
        # --execute without --landing-plan: exit 5 at the early checkpoint, nothing mutated, advice shown
        refs_before = B._git(self.probe, "ls-remote", str(self.origin)).stdout
        mut_before, batches_before = self.mutating_calls(), self.batch_files()
        rc, out = self.cli(True, ["--prs", "1,2,3"])
        ev("S11 execute refusal", rc=rc, advice_shown="recommends: group" in out, refs_unchanged=B._git(self.probe, "ls-remote", str(self.origin)).stdout == refs_before,
           mutating_calls=self.mutating_calls() == mut_before, batch_records_unchanged=self.batch_files() == batches_before)
        self.assertEqual(rc, merge_cleanup.EXIT_LANDING_PLAN_REQUIRED)
        self.assertIn("The dry run's landing forecast recommends: group", out)
        self.assertEqual(B._git(self.probe, "ls-remote", str(self.origin)).stdout, refs_before)
        self.assertEqual(self.mutating_calls(), mut_before)
        self.assertEqual(self.batch_files(), batches_before)
        # explicit group decision matching the forecast: recorded as a match, survives reload, renders in the issue block
        rc, out = self.cli(True, ["--prs", "1,2,3", "--landing-plan", "group", "--landing-reason", "manual check: three ledger-conflicting PRs"])
        files = self.batch_files()
        rec = json.loads((self.primary / ".tick" / "merge-cleanup" / files[0]).read_text())
        fc = rec["landing_plan"]["forecast"]
        block = batch_record.script_block(rec)
        ev("S11 execute group (match)", rc=rc, state=fc["state"], recommendation=fc["recommendation"], stale=fc["stale"],
           reason_is_operators=rec["landing_plan"]["reason"], rendered=[l for l in block.splitlines() if "Forecast (GH-1452)" in l])
        self.assertEqual((rc, fc["state"], fc["stale"]), (0, "match", False))
        self.assertEqual(rec["landing_plan"]["reason"], "manual check: three ledger-conflicting PRs")
        self.assertIn("Forecast (GH-1452)", block)
