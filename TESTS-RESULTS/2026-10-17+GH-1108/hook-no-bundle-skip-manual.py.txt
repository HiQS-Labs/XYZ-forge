#!/usr/bin/env python3
"""Manual control (GH-1108 review Blocker): the merge-cleanup Jog hook must leave a mapped
row alone when the merged tree carries no committed receipt bundle, and must still fail
when a bundle exists and `jog land` refuses.

Retained as a text record under TESTS-RESULTS (GH-831: no new suites). Run from the repo
root of the candidate:  python3 TESTS-RESULTS/2026-10-17+GH-1108/hook-no-bundle-skip-manual.py.txt
It builds a throwaway git repo in a temp dir, stubs `gh`, `jog list` and `jog land` through
module attributes, and never touches the primary or any ledger.
"""
import json, os, subprocess, sys, tempfile, types
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT / "skills/2-daily/merge-cleanup/scripts"))
import merge_cleanup as mc  # noqa: E402

RESULTS = []


def check(name, ok):
    RESULTS.append((name, ok))
    print(("PASS" if ok else "FAIL") + ": " + name)


def git(repo, *args):
    return subprocess.run(["git", "-C", str(repo), *args], check=True, capture_output=True, text=True)


def make_repo(with_bundle_for=None):
    repo = Path(tempfile.mkdtemp(prefix="gh1108-hook-"))
    git(repo, "init", "-q", "-b", "development")
    git(repo, "config", "user.email", "t@test.invalid")
    git(repo, "config", "user.name", "t")
    (repo / "releases.db").write_bytes(b"not-a-real-db")
    (repo / "releases.sql").write_text("-- table: jog_queue\nINSERT INTO jog_queue(x) VALUES(1);\n")
    (repo / "utils/py").mkdir(parents=True)
    (repo / "utils/py/releases_app.py").write_text("# stub\n")
    (repo / "utils/py/harness_paths.py").write_text("def require_primary(root):\n    return root\n"
                                                    "def github_slug_from_origin(root):\n    return None\n")
    if with_bundle_for:
        b = repo / f"TESTS-RESULTS/2026-10-17+GH-{with_bundle_for}/jog/gh{with_bundle_for}-exec1"
        b.mkdir(parents=True)
        (b / "receipt.json").write_text("{}\n")
    git(repo, "add", "-A")
    git(repo, "commit", "-q", "-m", "fixture")
    return repo


class Proc:
    def __init__(self, rc, out="", err=""):
        self.returncode, self.stdout, self.stderr = rc, out, err


def run_hook(repo, rows, land_rc):
    calls = []

    def fake_run(argv, **kw):
        if "list" in argv:
            return Proc(0, json.dumps(rows))
        if "land" in argv:
            calls.append(argv)
            return Proc(land_rc, "jog: landed" if land_rc == 0 else "", "jog: refusing to land")
        raise AssertionError(argv)

    mc.subprocess = types.SimpleNamespace(run=fake_run, TimeoutExpired=subprocess.TimeoutExpired)
    mc._gh = lambda args, cwd, timeout=180: Proc(0, json.dumps(
        {"number": 7, "title": "x", "body": "Closes #1075", "headRefName": "fix/XYZ-forge-gh1075-x"}))
    try:
        ok = mc.run_jog_post_merge_land(7, repo)
    finally:
        mc.subprocess = subprocess
    return ok, calls


rows = [{"gh_number": 1075, "status": "pending"}]

# A. mapped row, no bundle in the merged tree -> hook returns True and never calls jog land
repo_a = make_repo()
ok, calls = run_hook(repo_a, rows, land_rc=2)
check("A no-bundle mapped row: hook returns True", ok is True)
check("A no-bundle mapped row: jog land not invoked", calls == [])

# B. mapped row WITH bundle, jog land refuses -> hook returns False (landing stops)
repo_b = make_repo(with_bundle_for=1075)
ok, calls = run_hook(repo_b, rows, land_rc=2)
check("B bundle present + refusal: hook returns False", ok is False)
check("B bundle present + refusal: jog land invoked once", len(calls) == 1)

# C. mapped row WITH bundle, jog land succeeds -> hook returns True
ok, calls = run_hook(repo_b, rows, land_rc=0)
check("C bundle present + success: hook returns True", ok is True and len(calls) == 1)

# D. bad gh JSON -> named failure, no traceback
mc._gh = lambda args, cwd, timeout=180: Proc(0, "not json")
mc.subprocess = types.SimpleNamespace(run=lambda argv, **kw: Proc(0, json.dumps(rows)),
                                      TimeoutExpired=subprocess.TimeoutExpired)
try:
    ok = mc.run_jog_post_merge_land(7, repo_a)
    check("D non-JSON gh output: returns False without raising", ok is False)
finally:
    mc.subprocess = subprocess

bad = [n for n, ok in RESULTS if not ok]
print(f"{len(RESULTS) - len(bad)} pass, {len(bad)} fail")
sys.exit(1 if bad else 0)
