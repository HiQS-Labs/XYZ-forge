BASE="MARATHON-P1-TURN"

seed_relay() {  # <status> <recorded-task>
  local status="$1" task="$2"
  mkdir -p "$A/phases/p1"
  printf '# Marathon Phase p1\nSTATUS: %s\nNEXT: agy (Reviewer)\n\n<!-- marathon-drive: task=%s builder=codex reviewer=agy round-cap=5 -->\n\nbody\n' \
    "$status" "$task" > "$A/phases/p1/RELAY.md"
}

attest_seed() {  # <task> — GH-505: a hand-seeded Approved lane is satisfied only with the driver's record
  grep -q '^STATUS: Approved' "$A/phases/p1/RELAY.md" 2>/dev/null || return 0
  TICK_REPO_ROOT="$A" TICK_BIN="$TICK" bash "$ATTEST_STUB" --relay-file "$A/phases/p1/RELAY.md" --relay-task "$1" --reviewer agy --target-root "$A" >/dev/null 2>&1
}

mk_token() {  # <task> <done|claimed>
  tick_a log task.created "$1" --agent marathon >/dev/null 2>&1 || true
  tick_a claim "$1" --agent codex --paths "phases/p1/RELAY.md" >/dev/null 2>&1 || true
  [ "$2" = "done" ] && { tick_a done "$1" --agent codex >/dev/null 2>&1 || true; }
}

run_driver() {  # [extra marathon-drive args...]
  : > "$RD_CALLS"
  MARATHON_ROOT="$A" MARATHON_RELAY_DRIVE="$RD" TICK_REPO_ROOT="$A" TICK_BIN="$TICK" \
  CODEX_BIN="$STUB_CODEX" AGY_BIN="$STUB_AGY" \
  python3 "$DRIVER" --phases-dir "$A/phases" --phase-brief "$BRIEF" \
    --reviewer agy --builder codex --pre-advance-cmd "true" "$@" 2>&1
}

reset_state() { rm -rf "$A/phases" "$A/.tick"; tick_a init >/dev/null 2>&1 || true; : > "$RD_CALLS"; }


seed_relay "Approved" "$BASE"

 archive-failure) printf blocked > "$A/.tick/attempt-history"; run_driver --result-file "$WORK/result.json" > "$WORK/driver.log" 2>&1; rc=$? ;;
 *) run_driver --result-file "$WORK/result.json" > "$WORK/driver.log" 2>&1; rc=$? ;;
esac
python3 - "$A" "$WORK" "$rc" "$CASE" "$MODE" "$OUT" <<'CHECK'
