#!/usr/bin/env bash
# GH-833 witnesses, v2 (GH-844): V1 + V1b on the tree, then red controls, each in a new `mktemp -d` copy of the
# 14 files; nothing is deleted. v2 returns an AGGREGATE status: it exits non-zero if a green check fails, a red
# control passes, or a mutation does not apply. WITNESS_SELFTEST=1 adds a control that must break the aggregate.
# Usage: witness-script-v2.sh <repo root> <dir holding prs-order-check.sh and prs-definition-check-v2.py>
set -u
REPO="$1" CHK="$2" STATUS=0
FILES="HOW-TO-USE.md ROUTER.md AGENTS.md PROJECT/PDDA.md SOP.md ARCHITECTURE.md RELEASES-DB-FAQS.md README.md skills/2-daily/releases/SKILL.md skills/1-hourly/start-task/SKILL.md skills/2-daily/merge-cleanup/SKILL.md skills/2-daily/start-marathon/SKILL.md skills/3-weekly/10days/SKILL.md skills/3-weekly/end-of-week/SKILL.md"
fresh() {  # sets W to a NEW scratch dir holding the 14 files
  W="$(mktemp -d "${TMPDIR:-/tmp}/gh833-wit.XXXXXX")" && [ -d "$W" ] || { echo "mktemp failed" >&2; exit 2; }
  (cd "$REPO" && tar -cf - $FILES) | tar -xf - -C "$W"
}
v1()   { bash "$CHK/prs-order-check.sh" "$1"; }
v1b()  { python3 "$CHK/prs-definition-check-v2.py" "$1"; }
v1b1() { python3 "$CHK/prs-definition-check.py" "$1"; }   # v1, only to show the gap v2 closes
green() {  # <name> <check fn>
  echo "## green: $1 on the tree"; "$2" "$REPO"; local rc=$?
  if [ "$rc" -eq 0 ]; then echo "rc=0 (want 0) ok"; else echo "rc=$rc (want 0) UNEXPECTED"; STATUS=1; fi
}
ctl() {  # <name> <check fn> <want rc> <python mutation over the scratch copy>
  fresh
  if ! python3 - "$W" <<<"$4"; then echo; echo "## red control: $1"; echo "MUTATION DID NOT APPLY"; STATUS=1; return; fi
  echo; echo "## control: $1"; "$2" "$W" | grep -E "FAIL|pass"; local rc=${PIPESTATUS[0]}
  if [ "$rc" -eq "$3" ]; then echo "rc=$rc (want $3) ok"; else echo "rc=$rc (want $3) UNEXPECTED"; STATUS=1; fi
}
echo "# tree: $(git -C "$REPO" rev-parse --short HEAD)"
green V1 v1; green V1b-v2 v1b
ctl "remove ROUTER's trinity line (V1)" v1 1 'import sys,re;p=sys.argv[1]+"/ROUTER.md";s=open(p).read();n=re.sub(r"\nXYZ Forge is three parts:[^\n]*\n","\n",s);assert n!=s;open(p,"w").write(n)'
ctl "delete the glossary entry (V1b v2)" v1b 1 'import sys,re;p=sys.argv[1]+"/HOW-TO-USE.md";s=open(p).read();n=re.sub(r"\n- \*\*PRS\*\*.*?(?=\n\n)","",s,flags=re.S);assert n!=s;open(p,"w").write(n)'
ctl "break ROUTER's glossary anchor (V1b v2)" v1b 1 'import sys;p=sys.argv[1]+"/ROUTER.md";s=open(p).read();n=s.replace("HOW-TO-USE.md#glossary--the-five-terms","HOW-TO-USE.md#glossary--the-four-terms");assert n!=s;open(p,"w").write(n)'
ctl "restate the entry in RELEASES-DB-FAQS.md (V1b v2)" v1b 1 'import sys;p=sys.argv[1]+"/RELEASES-DB-FAQS.md";s=open(p).read();open(p,"w").write(s+"\n- **PRS** — the Product Release System.\n")'
ctl "break one skill URL anchor (V1b v2)" v1b 1 'import sys;p=sys.argv[1]+"/skills/3-weekly/10days/SKILL.md";s=open(p).read();n=s.replace("#glossary--the-five-terms-youll-hit-first","#glossary");assert n!=s;open(p,"w").write(n)'
MOVE='import sys,re;p=sys.argv[1]+"/HOW-TO-USE.md";s=open(p).read();m=re.search(r"\n(- \*\*PRS\*\*.*?)(?=\n\n)",s,flags=re.S);assert m;e=m.group(1);s=s[:m.start()]+s[m.end():];i=s.index("\n## FAQ\n")+len("\n## FAQ\n");s=s[:i]+"\n"+e+"\n"+s[i:];open(p,"w").write(s)'
ctl "move the entry under ## FAQ (V1b v2; new in GH-844)" v1b 1 "$MOVE"
ctl "the same move under v1 (the gap GH-844 closes: v1 passes)" v1b1 0 "$MOVE"
if [ "${WITNESS_SELFTEST:-0}" = 1 ]; then
  ctl "SELFTEST: a no-op mutation expected to fail V1 (must break the aggregate)" v1 1 'import sys;pass'
fi
echo; echo "AGGREGATE: $([ "$STATUS" -eq 0 ] && echo PASS || echo FAIL)"
exit "$STATUS"
