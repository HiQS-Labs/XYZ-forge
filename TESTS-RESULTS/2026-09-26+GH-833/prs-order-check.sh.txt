#!/usr/bin/env bash
# GH-833 V1: in each file, the first line using PRS as a word must also spell out "Product Release System".
# Usage: prs-order-check.sh <tree root>   Exit 0 = all pass; 1 = any file fails (missing, or bare PRS first).
set -u
ROOT="${1:?tree root}"
FILES="HOW-TO-USE.md ROUTER.md AGENTS.md PROJECT/PDDA.md SOP.md ARCHITECTURE.md RELEASES-DB-FAQS.md README.md
skills/2-daily/releases/SKILL.md skills/1-hourly/start-task/SKILL.md skills/2-daily/merge-cleanup/SKILL.md
skills/2-daily/start-marathon/SKILL.md skills/3-weekly/10days/SKILL.md skills/3-weekly/end-of-week/SKILL.md"
fail=0 n=0
for f in $FILES; do
  n=$((n + 1))
  if [ ! -f "$ROOT/$f" ]; then echo "FAIL $f: file missing"; fail=$((fail + 1)); continue; fi
  first=$(grep -nw PRS "$ROOT/$f" | head -1)
  if [ -z "$first" ]; then echo "FAIL $f: no PRS at all"; fail=$((fail + 1))
  elif printf '%s' "$first" | grep -q "Product Release System"; then echo "ok   $f:${first%%:*}"
  else echo "FAIL $f:${first%%:*}: bare PRS before its definition"; fail=$((fail + 1)); fi
done
echo "V1: $((n - fail))/$n pass"
[ "$fail" -eq 0 ]
