#!/usr/bin/env python3
"""GH-851 / GH-852 witnesses (plan steps F1-F5). Manual checks, not a registered suite (AGENTS.md: no new tests).

One script, run unchanged against a base checkout and a head checkout:

    python3 witness.py.txt --checkout <clone> --label base|head --out <dir>

Each case runs in its own process group under a 10 s outer watchdog; a case still running when the
watchdog fires is recorded as BLOCKED. Every case writes one JSON line to <out>/<label>.jsonl and
its full stdout/stderr to <out>/<label>-<case>.log.
"""
import argparse, contextlib, io, json, os, signal, subprocess, sys, tempfile, time
from pathlib import Path
from unittest import mock

WATCHDOG_S = 10
H, M, P_NEW, P_OLD = "a" * 40, "b" * 40, "c" * 40, "d" * 40


def _load(checkout):
    sys.path.insert(0, str(Path(checkout) / "skills/2-daily/merge-cleanup/scripts"))
    import merge_cleanup, scan_clones
    return merge_cleanup, scan_clones


def _stub_git(tmp, stall_on):
    """A `git` first on PATH that sleeps (exec, so a kill reaches it) for one subcommand only."""
    real = subprocess.run(["which", "git"], capture_output=True, text=True).stdout.strip()
    d = Path(tmp) / "stubbin"; d.mkdir()
    (d / "git").write_text(f'#!/bin/bash\nfor a in "$@"; do [ "$a" = "{stall_on}" ] && exec sleep 600; done\nexec "{real}" "$@"\n')
    (d / "git").chmod(0o755)
    os.environ["PATH"] = f"{d}:{os.environ['PATH']}"


def _repo(tmp, name="primary"):
    r = Path(tmp) / name
    subprocess.run(["git", "init", "-q", "-b", "development", str(r)], check=True)
    subprocess.run(["git", "-C", str(r), "-c", "user.name=w", "-c", "user.email=w@e", "commit", "-q",
                    "--allow-empty", "-m", "p"], check=True)
    subprocess.run(["git", "-C", str(r), "remote", "add", "origin", str(Path(tmp) / "nowhere.git")], check=True)
    return r


def _shorten(mc_or_sc):
    real = mc_or_sc.run_git
    def wrap(cwd, args, timeout=None, **kw):  # the plan's C2/D2 wrapper: None stays None
        return real(cwd, args, timeout=None if timeout is None else min(timeout, 2), **kw)
    return wrap


def _verdict(primary):
    return {"path": str(primary), "integration_branch": "development", "current_branch": "development",
            "is_clean": True, "dirty_count": 0, "on_integration_branch": True, "unpushed_on_integration": 0,
            "can_ff": True, "operation_in_progress": "", "evidence_complete": True, "landing_ready": True,
            "blockers": []}

def w1_clone_stall(mc, sc, tmp):
    """F1: a stalled `clone` in validate_head_in_second_clone, bound shortened through the wrapper."""
    _stub_git(tmp, "clone")
    primary = _repo(tmp)
    with mock.patch.object(mc, "run_git", side_effect=_shorten(mc)):
        ok, why = mc.validate_head_in_second_clone(primary, primary, H, Path(tmp))

def w2_timeouts(mc, sc, tmp):
    """F1: the `timeout` each of the six network git sites passes (None = unbounded)."""
    seen = {}
    real_mc, real_sc = mc.run_git, sc.run_git
    def rec(site, real=None):
        def fn(cwd, args, timeout=None, **kw):
            if args[0] not in ("clone", "push", "fetch"):  # local queries pass through
                return (real or real_mc)(cwd, args, timeout=timeout)
            seen.setdefault(site, []).append([args[0], timeout])
            return subprocess.CompletedProcess(args, 1, "", "witness stop")
        return fn
    primary = _repo(tmp)
    with mock.patch.object(mc, "run_git", side_effect=rec("merge_cleanup.py:316 clone")):
        mc.validate_head_in_second_clone(primary, primary, H, Path(tmp))
    pr = {"number": 7, "headRefOid": H, "headRefName": "feat/x"}
    with mock.patch.object(mc, "refresh_pr", return_value=dict(pr)), \
         mock.patch.object(mc, "run_git", side_effect=rec("merge_cleanup.py:343 push")):
        mc.push_resolved_head(pr, primary, H, primary)
    def reconcile_git(cwd, args, timeout=None, **kw):
        if args[:2] == ["rev-parse", "HEAD"]:
            return subprocess.CompletedProcess(args, 0, M + "\n", "")
        return rec("merge_cleanup.py:550 fetch")(cwd, args, timeout=timeout)
    with mock.patch.object(mc, "wait_for_hosted_reconcile", return_value="success"), \
         mock.patch.object(mc, "run_git", side_effect=reconcile_git):
        mc.run_post_merge_reconcile(7, primary, "development", dry_run=False)
    def p5_git(cwd, args, timeout=None, **kw):
        if args[0] == "status":
            return subprocess.CompletedProcess(args, 0, "", "")
        site = {"push": "merge_cleanup.py:601 push", "fetch": "merge_cleanup.py:605 fetch"}.get(args[0], "other " + args[0])
        seen.setdefault(site, []).append([args[0], timeout])
        return subprocess.CompletedProcess(args, 0 if args[0] == "push" else 1, "", "witness stop")
    with mock.patch.object(mc, "run_git", side_effect=p5_git):
        mc.commit_and_push_phase5_writes(7, primary, "development")
    with mock.patch.object(sc, "run_git", side_effect=rec("scan_clones.py:437 fetch", real_sc)):
        sc.classify_local_refs(primary, "development")
    return {"timeouts": seen}

def w3_env(mc, sc, tmp):
    """F1: after main() has set up, the child-git env holds the low-speed abort; an operator value wins."""
    res = {}
    for mod, name in ((mc, "merge_cleanup"), (sc, "scan_clones")):
        for preset in (None, "5"):
            for k in ("GIT_HTTP_LOW_SPEED_LIMIT", "GIT_HTTP_LOW_SPEED_TIME"):
                os.environ.pop(k, None)
            if preset:
                os.environ["GIT_HTTP_LOW_SPEED_LIMIT"] = preset
            # an argv argparse rejects: main() returns through SystemExit right after its setup
            with mock.patch.object(sys, "argv", [name + ".py", "--no-such-flag"]), \
                 contextlib.redirect_stderr(io.StringIO()):
                try:
                    mod.main()
                except SystemExit:
                    pass
            res[f"{name} preset={preset}"] = {k: os.environ.get(k) for k in ("GIT_HTTP_LOW_SPEED_LIMIT", "GIT_HTTP_LOW_SPEED_TIME")}
    return res

def _gh_stub(view, runs=None, merge_rc=1):
    calls = []
    def fn(args, cwd, timeout=180):
        calls.append(list(args[:2]))
        if args[:2] == ["pr", "merge"]:
            return subprocess.CompletedProcess(args, merge_rc, "", "gh: killed / network error")
        if args[:2] == ["pr", "view"]:
            if view is None:
                return subprocess.CompletedProcess(args, 1, "", "HTTP 404: Not Found")
            return subprocess.CompletedProcess(args, 0, json.dumps(view), "")
        if args[:2] == ["run", "list"]:
            return subprocess.CompletedProcess(args, 0, json.dumps(runs or []), "")
        return subprocess.CompletedProcess(args, 1, "", "unexpected " + " ".join(args))
    return fn, calls

def _f3(mc, sc, tmp, view, red_control=False):
    primary = _repo(tmp)
    phead = subprocess.run(["git", "-C", str(primary), "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip()
    runs = [{"databaseId": 99, "status": "in_progress", "conclusion": None, "headSha": H, "event": "pull_request"}]
    fn, calls = _gh_stub(view, runs)
    os.environ.update({"MERGE_CLEANUP_HOSTED_WAIT_S": "1", "MERGE_CLEANUP_HOSTED_POLL_S": "1",
                       "MERGE_CLEANUP_HOSTED_GRACE_S": "1"})
    reconcile_calls, local_calls, looked_for = [], [], []
    real_reconcile, real_wait = mc.run_post_merge_reconcile, mc.wait_for_hosted_reconcile
    def spy_reconcile(*a, **kw):
        reconcile_calls.append({k: kw.get(k) for k in ("pr_head", "merged_head")})
        return real_reconcile(*a, **kw)
    def spy_wait(merged_head, *a, **kw):
        looked_for.append({"merged_head": merged_head, "pr_head": kw.get("pr_head")})
        if red_control:  # deliberate mutation: the local writer runs while the run on H is active
            mc.run_local_wave_reconcile(42, primary)
        return real_wait(merged_head, *a, **kw)
    def local(pr_num, repo):
        local_calls.append(pr_num); return False
    with mock.patch.object(sys, "argv", ["merge_cleanup.py", "--primary", str(primary), "--reconcile-pr", "42", "--execute"]), \
         mock.patch.object(mc, "inspect_primary_landing", return_value=_verdict(primary)), \
         mock.patch.object(mc, "_gh", side_effect=fn), \
         mock.patch.object(mc, "run_post_merge_reconcile", side_effect=spy_reconcile), \
         mock.patch.object(mc, "wait_for_hosted_reconcile", side_effect=spy_wait), \
         mock.patch.object(mc, "run_local_wave_reconcile", side_effect=local), \
         mock.patch.object(mc, "_sleep", lambda s: None):
        rc = mc.main()
    return {"rc": rc, "primary_head": phead, "pr_head_H": H, "merge_M": M, "looked_for": looked_for,
            "run_post_merge_reconcile_calls": reconcile_calls, "local_writer_calls": len(local_calls)}

def r3_clone_retry_after_stall(mc, sc, tmp):
    """#820 landing, 2026-09-27: the landing clone stalls past its bound after creating its directory;
    the GH-623 retry must not then fail on 'destination path already exists'."""
    real = subprocess.run(["which", "git"], capture_output=True, text=True).stdout.strip()
    bare = Path(tmp) / "origin.git"
    subprocess.run(["git", "init", "-q", "--bare", "-b", "development", str(bare)], check=True)
    seed = _repo(tmp, "seed")
    subprocess.run(["git", "-C", str(seed), "push", "-q", str(bare), "HEAD:refs/heads/development"], check=True)
    primary = _repo(tmp)
    subprocess.run(["git", "-C", str(primary), "remote", "set-url", "origin", str(bare)], check=True)
    d = Path(tmp) / "stubbin"; d.mkdir(); flag = Path(tmp) / "first-clone-done"
    (d / "git").write_text(f'''#!/bin/bash
for a in "$@"; do if [ "$a" = clone ] && [ ! -e "{flag}" ]; then
  touch "{flag}"; dest="${{@: -1}}"; mkdir -p "$dest/.git"; echo partial > "$dest/.git/HEAD"; exec sleep 600
fi; done
exec "{real}" "$@"
''')
    (d / "git").chmod(0o755)
    os.environ["PATH"] = f"{d}:{os.environ['PATH']}"
    pr = {"number": 820, "headRefOid": H, "headRefName": "feat/x"}
    with mock.patch.object(mc, "run_git", side_effect=_shorten(mc)), mock.patch.object(mc, "_sleep", lambda s: None):

    err = out.get("error") or ""
    return {"error": err[:200], "clone_retry_failed_on_leftover_dir": "already exists" in err,
            "clone_succeeded_on_retry": not err.startswith("git clone failed")}

CASES = [w1_clone_stall, w2_timeouts, w2a_scan_fetch_stall, w3_env, w4_transient, f2_merge_recovery,
         f2_control_open, f3_lookup, f3_open_refused, f3_view_error, f3_red_control, f4_regate, f5_default,

def run_one(checkout, case, result_path):
    mc, sc = _load(checkout)
    with tempfile.TemporaryDirectory() as tmp:
        out = globals()[case](mc, sc, tmp)
    Path(result_path).write_text(json.dumps(out, default=str))


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--checkout", required=True)
    ap.add_argument("--label", required=True)
    ap.add_argument("--out", required=True)
    ap.add_argument("--one"), ap.add_argument("--result")
    ap.add_argument("--cases", help="comma-separated subset of case names")
    a = ap.parse_args()
    if a.one:
        return run_one(a.checkout, a.one, a.result)
    out = Path(a.out); out.mkdir(parents=True, exist_ok=True)
    sha = subprocess.run(["git", "-C", a.checkout, "rev-parse", "HEAD"], capture_output=True, text=True).stdout.strip()
    with open(out / f"{a.label}.jsonl", "w") as jl:
        for case in CASES:
            name = case.__name__
            if a.cases and name not in a.cases.split(","):
                continue
            res_file = out / f".{a.label}-{name}.json"
            log_file = out / f"{a.label}-{name}.log"
            t0 = time.time()
            with open(log_file, "w") as lf:
                proc = subprocess.Popen([sys.executable, __file__, "--checkout", a.checkout, "--label", a.label,
                                         "--out", a.out, "--one", name, "--result", str(res_file)],
                                        stdout=lf, stderr=subprocess.STDOUT, start_new_session=True)
                try:
                    proc.wait(timeout=WATCHDOG_S)
                    outcome = "returned" if proc.returncode == 0 else f"crashed rc={proc.returncode}"
                except subprocess.TimeoutExpired:
                    os.killpg(proc.pid, signal.SIGKILL); proc.wait()
                    outcome = f"BLOCKED at the {WATCHDOG_S}s outer watchdog"
            result = json.loads(res_file.read_text()) if res_file.exists() else None
            res_file.unlink(missing_ok=True)
            if log_file.stat().st_size == 0:
                log_file.write_text(f"(case {name} printed nothing; outcome: {outcome})\n")
            line = {"label": a.label, "sha": sha, "case": name, "outcome": outcome,
                    "elapsed_s": round(time.time() - t0, 2), "result": result}
            jl.write(json.dumps(line, default=str) + "\n")
            print(json.dumps(line, default=str)[:400])


if __name__ == "__main__":
    main()
