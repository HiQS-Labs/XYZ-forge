#!/usr/bin/env python3
"""GH-1453 manual checks A1-A14 against a local fixture (no suite, GH-831).

Usage: fx.py <cand|base> <rev> <source-clone> <workdir> [check ...]

Each check builds its own fixture: a bare origin whose `development` is <rev>, a primary clone
carrying the primary pin, a publication clone, and an `other` clone that plays another session
(pinned as its own primary so it can also write Jog rows). The code under test is the primary's
checkout of <rev>. Ledger writes use the supported CLI and the canonical writers only.
XYZ_WORK_CONNECTORS=0 keeps every write local. Exit 0 only if every selected check gives the
verdict expected for the mode (cand: behaviour present; base: the stated red control).
"""
import json
import os
import shutil
import sqlite3
import subprocess
import sys
import tempfile
from pathlib import Path

MODE, REV, SOURCE, WORK = sys.argv[1], sys.argv[2], sys.argv[3], Path(sys.argv[4]).resolve()
SELECTED = sys.argv[5:]
ENV = {**os.environ, "XYZ_WORK_CONNECTORS": "0", "GIT_TERMINAL_PROMPT": "0"}
N = 1315          # the queue row the wake writes (pending at <rev>)
B_CANDIDATES = (1448, 1414, 1356, 1360)
RESULTS = []

PRELUDE = """
import os, sys, json
ROOT = os.environ['FX_ROOT']; PUB = os.environ['FX_PUB']; N = int(os.environ['FX_N'])
sys.path.insert(0, ROOT + '/utils/py')
import jog_run
from jog_run import JogCheckpoint, ContractError
from releases_app import jog_acquire_lease, jog_set_status
"""


def run(cmd, cwd=None, ok=True, env=None):
    p = subprocess.run(cmd, cwd=cwd, capture_output=True, text=True, env=env or ENV)
    out = (p.stdout + p.stderr).strip()
    if ok and p.returncode:
        raise SystemExit("FIXTURE ERROR rc=%s: %s\n%s" % (p.returncode, " ".join(map(str, cmd)), out))
    return p.returncode, out


def git(repo, *args, ok=True):
    return run(["git", "-C", str(repo), *args], ok=ok)[1] if ok else run(["git", "-C", str(repo), *args], ok=False)


def triple(db, gh):
    conn = sqlite3.connect(Path(db).resolve().as_uri() + "?mode=ro", uri=True)
    try:
        row = conn.execute("SELECT status, attempt_count, failure_reason FROM jog_queue WHERE gh_number=?",
                           (gh,)).fetchone()
        return list(row) if row else None
    finally:
        conn.close()


def receipts(db):
    conn = sqlite3.connect(Path(db).resolve().as_uri() + "?mode=ro", uri=True)
    try:
        return conn.execute("SELECT count(*) FROM op_receipts").fetchone()[0]
    finally:
        conn.close()


class Fx:
    def __init__(self, name, view=False, with_b=False):
        self.dir = WORK / MODE / name
        if self.dir.exists():
            shutil.rmtree(self.dir)
        self.dir.mkdir(parents=True)
        self.remote = self.dir / "remote.git"
        run(["git", "clone", "-q", "--bare", SOURCE, str(self.remote)])
        git(self.remote, "update-ref", "refs/heads/development", REV)
        git(self.remote, "symbolic-ref", "HEAD", "refs/heads/development")
        self.primary, self.pub, self.other = (self.dir / n for n in ("primary", "pub", "other"))
        for clone in (self.primary, self.pub, self.other):
            run(["git", "clone", "-q", str(self.remote), str(clone)])
        key = run([sys.executable, "-c",
                   "import sys;sys.path.insert(0,'utils/py');import harness_paths as h;print(h.primary_config_key('.'))"],
                  cwd=self.primary)[1].strip()
        for clone in (self.primary, self.other):
            git(clone, "config", key, str(clone))   # LOCAL config only; never global
        self.b = None
        seed = []
        if view:
            run([sys.executable, "utils/timeline/export_timeline.py", "--db", "releases.db",
                 "--preview", "RELEASES-PREVIEW.html"], cwd=self.primary)
            seed.append("RELEASES-PREVIEW.html")
        if with_b:
            for cand in B_CANDIDATES:
                if self.app(self.primary, "jog", "add", str(cand), ok=False)[0] == 0:
                    self.b = cand
                    break
            assert self.b, "no second queue row could be added"
            seed += ["releases.db", "releases.sql"]
        if seed:
            git(self.primary, "add", "--", *seed)
            git(self.primary, "commit", "-q", "-m", "fixture seed")
            git(self.primary, "push", "-q", "origin", "development")
            for clone in (self.pub, self.other):
                git(clone, "pull", "-q", "--ff-only")
        assert triple(self.primary / "releases.db", N) == ["pending", 0, None], "seed row GH-%d not pending" % N
        self.base = git(self.primary, "rev-parse", "HEAD")

    def app(self, repo, *args, ok=True):
        return run([sys.executable, str(repo / "utils/py/releases_app.py"), "--root", str(repo), *args],
                   cwd=repo, ok=ok)

    def py(self, code, root=None, ok=False):
        env = {**ENV, "FX_ROOT": str(root or self.primary), "FX_PUB": str(self.pub), "FX_N": str(N)}
        return run([sys.executable, "-c", PRELUDE + code], cwd=root or self.primary, ok=ok, env=env)

    def begin(self, write=True, ok=False):
        code = "c = JogCheckpoint(ROOT, PUB)\nassert c.begin()\n"
        if write:
            code += ("jog_acquire_lease(ROOT, N, os.getpid())\n"
                     "jog_set_status(ROOT, N, 'parked', failure_reason='fixture park')\n")
        return self.py(code, ok=ok)

    def finish(self):
        return self.py("c = JogCheckpoint(ROOT, PUB)\nc.state = json.loads(c.journal.read_text())\nc.finish()\n")

    def resume(self):
        """What a rerun of the same command does first: begin() on the retained journal."""
        return self.py("c = JogCheckpoint(ROOT, PUB)\nprint('BEGIN', c.begin())\n")

    def other_sync(self):
        git(self.other, "pull", "-q", "--ff-only")

    def other_commit(self, message, paths=("releases.db", "releases.sql"), push=True):
        git(self.other, "add", "--", *paths)
        git(self.other, "commit", "-q", "-m", message)
        if push:
            git(self.other, "push", "-q", "origin", "development")
        return git(self.other, "rev-parse", "HEAD")

    def other_roadmap(self, pri, push=True):
        """A disjoint ledger write by another session: re-rate GH-N's ROADMAP row (not its queue row)."""
        self.other_sync()
        self.app(self.other, "roadmap", "rate", "--issue-num", str(N), "--rated", "%d/41/50/51" % pri, "--force")
        return self.other_commit("other session: re-rate GH-%d pri %d" % (N, pri), push=push)

    def other_queue(self, gh, status, reason=None, lease=False):
        self.other_sync()
        code = ("jog_acquire_lease(ROOT, %d, 999999)\n" % gh if lease else
                "jog_set_status(ROOT, %d, %r, failure_reason=%r)\n" % (gh, status, reason))
        self.py(code, root=self.other, ok=True)
        return self.other_commit("other coordinator: GH-%d %s" % (gh, "lease" if lease else status))

    def remote_head(self):
        return git(self.remote, "rev-parse", "development")

    def remote_db(self):
        tmp = Path(tempfile.mkdtemp(dir=self.dir)) / "remote.db"
        tmp.write_bytes(subprocess.run(["git", "-C", str(self.remote), "show", "development:releases.db"],
                                       capture_output=True, check=True).stdout)
        return tmp

    def remote_pri(self):
        rows = json.loads(self.app(self.other, "roadmap", "list", "--json")[1])
        rows = rows if isinstance(rows, list) else rows.get("items", rows.get("rows", []))
        return [r.get("rating_pri") for r in rows if str(r.get("gh_number")) == str(N)]

    def journal(self):
        return json.loads((self.primary / ".tick/jog/checkpoint.json").read_text())

    def anc(self, a, b):
        return run(["git", "-C", str(self.remote), "merge-base", "--is-ancestor", a, b], ok=False)[0] == 0

    def clean(self, repo=None):
        return git(repo or self.primary, "status", "--porcelain", "--untracked-files=all") == ""

    def merging(self):
        return run(["git", "-C", str(self.primary), "rev-parse", "-q", "--verify", "MERGE_HEAD"], ok=False)[0] == 0

    def check_clean(self):
        return self.app(self.primary, "check", ok=False)[0] == 0


def record(name, verdict, **obs):
    RESULTS.append((name, verdict))
    print("\n=== %s [%s] %s" % (name, MODE, "PASS" if verdict else "FAIL"))
    for k, v in obs.items():
        text = str(v)
        print("  %s: %s" % (k, text if len(text) < 700 else text[:340] + " ... " + text[-340:]))


def published_state(fx, remote_commits):
    """The A1 result: wake's queue values and the remote's rows both published, cleanly."""
    head = fx.remote_head()
    return {
        "journal_phase": fx.journal().get("phase"),
        "remote_queue_row": triple(fx.remote_db(), N),
        "ancestors_ok": all(fx.anc(c, head) for c in remote_commits),
        "check_clean": fx.check_clean(),
        "primary_clean": fx.clean(),
        "primary_equals_remote": git(fx.primary, "rev-parse", "HEAD") == head,
        "no_backup": not (fx.primary / "releases.db.bak").exists(),
    }


def good(state):
    return (state["journal_phase"] == "published" and state["remote_queue_row"] == ["parked", 1, "fixture park"]
            and state["ancestors_ok"] and state["check_clean"] and state["primary_clean"]
            and state["primary_equals_remote"] and state["no_backup"])

           rc=rc, out=out[-300:], **state)

def a3_a5():
    fx = Fx("a3")
    fx.begin(ok=True)
    x = fx.other_queue(N, "parked", "other coordinator parked it")
    rc, out = fx.finish()
    source = fx.journal().get("source")
    restored = {
        "names_row": "GH-%d" % N in out and "same Jog queue row" in out,
        "head_is_checkpoint_commit": git(fx.primary, "rev-parse", "HEAD") == source,
        "no_merge_in_progress": not fx.merging(),
        "tree_clean_db_matches_commit": fx.clean(),
        "no_backup": not (fx.primary / "releases.db.bak").exists(),
        "remote_unchanged": fx.remote_head() == x,
        "journal_phase": fx.journal().get("phase"),
        "check_clean": fx.check_clean(),
    }
    # The command line: a rerun with the journal committed goes straight to publication.
    exe = fx.dir / "exec"
    run(["git", "clone", "-q", str(fx.remote), str(exe)])
    rc_cli, out_cli = fx.app(fx.primary, "jog", "run", "--checkpoint-root", str(fx.pub), "--max-tasks", "1",
                             "--target-root", str(exe), "--builder", "agy", "--reviewer", "codex", ok=False)
    cli = {"cli_rc": rc_cli, "cli_message": "jog: checkpoint held:" in out_cli,
           "cli_no_traceback": "Traceback" not in out_cli, "cli_tail": out_cli[-260:]}
    record("A3 overlap and restore", rc != 0 and all(v for k, v in restored.items() if k != "journal_phase")
           and restored["journal_phase"] == "committed" and rc_cli == 2 and cli["cli_message"] and cli["cli_no_traceback"],
           rc=rc, tail=out[-260:], **restored, **cli)
    # A5: the other session restores the row to its start values; the same command is rerun.
    x2 = fx.other_queue(N, "pending", None)
    before = receipts(fx.primary / "releases.db")
    rc5, out5 = fx.app(fx.primary, "jog", "run", "--checkpoint-root", str(fx.pub), "--max-tasks", "1",
                       "--target-root", str(exe), "--builder", "agy", "--reviewer", "codex", ok=False)
    state = published_state(fx, [x, x2, source])
    record("A5 rerun after a stop", rc5 == 0 and good(state) and "processing GH-" not in out5,
           rc=rc5, out=out5[-300:], no_dispatch="processing GH-" not in out5,
           receipts_before=before, **state)

def a13():
    """A run dies while its own reconcile merge is open; the rerun restores it and publishes."""
    fx = Fx("a13")
    fx.begin(ok=True)
    x = fx.other_roadmap(67)
    rc, out = fx.py(
        "c = JogCheckpoint(ROOT, PUB)\nc.state = json.loads(c.journal.read_text())\n"
        "def killed(live):\n    raise KeyboardInterrupt('fixture: process killed mid-merge')\n"
        "c._replay = killed\nc.finish()\n")
    j = fx.journal()
    died = {"rc": rc, "merge_left_open": fx.merging(), "journal_records_merge_target": j.get("merging") == x,
            "journal_phase_when_killed": j.get("phase")}
    rc2, out2 = fx.resume()
    state = published_state(fx, [x, j.get("source")])
    record("A13 own interrupted merge is restored on rerun",
           rc != 0 and died["merge_left_open"] and died["journal_records_merge_target"] and rc2 == 0
           and good(state) and "merging" not in fx.journal(),
           **died, rerun_rc=rc2, rerun_out=out2[-240:], **state)


def a14():
    """A merge the operator opened is never aborted: the rerun refuses and leaves it intact."""
    fx = Fx("a14")
    fx.begin(ok=True)
    hook = fx.pub / ".git/hooks/pre-push"
    hook.write_text("#!/bin/sh\nexit 1\n")
    hook.chmod(0o755)
    rc, out = fx.finish()                               # checkpoint committed; the push is refused
    hook.unlink()
    source = fx.journal().get("source")
    fx.other_sync()
    (fx.other / "operator-repair.txt").write_text("operator work in progress\n")
    y = fx.other_commit("operator: repair commit", paths=("operator-repair.txt",), push=False)
    git(fx.primary, "fetch", "-q", str(fx.other), "development")
    run(["git", "-C", str(fx.primary), "merge", "--no-commit", "--no-ff", y], ok=False)
    opened = fx.merging() and git(fx.primary, "rev-parse", "MERGE_HEAD") == y
    rc2, out2 = fx.resume()
    kept = fx.merging() and git(fx.primary, "rev-parse", "MERGE_HEAD") == y \
        and (fx.primary / "operator-repair.txt").exists()
    record("A14 a merge this wake did not start is preserved",
           rc != 0 and opened and rc2 != 0 and kept
           and "checkpoint refuses a merge in progress that this wake did not start" in out2
           and git(fx.primary, "rev-parse", "HEAD") == source and fx.journal().get("phase") == "committed",
           first_rc=rc, operator_merge_open=opened, rerun_rc=rc2, rerun_tail=out2[-200:],
           operator_merge_still_open_with_its_file=kept,
           head_is_checkpoint_commit=git(fx.primary, "rev-parse", "HEAD") == source,
           journal_phase=fx.journal().get("phase"))
